import requests
import json

# URLs for UAT and PROD
uat_url = "http://uat-kibana-url.com/app/discover#/?g=(filters:!(),...)"
prod_url = "http://prod-kibana-url.com/app/discover#/?g=(filters:!(),...)"

# Function to fetch data from the URL
def fetch_data(url):
    headers = {"Content-Type": "application/json"}
    try:
        response = requests.get(url, headers=headers)
        if response.status_code == 200:
            return response.json()  # Adjust parsing depending on Kibana response
        else:
            print(f"Failed to fetch data from {url}. Status code: {response.status_code}")
            return None
    except Exception as e:
        print(f"Error fetching data: {e}")
        return None

# Fetch data from UAT and PROD
uat_data = fetch_data(uat_url)
prod_data = fetch_data(prod_url)

# Compare data if both responses are successful
if uat_data and prod_data:
    # Assuming 'hits' contains the document data in Elasticsearch responses
    uat_docs = {doc["_id"]: doc["_source"] for doc in uat_data["hits"]["hits"]}
    prod_docs = {doc["_id"]: doc["_source"] for doc in prod_data["hits"]["hits"]}

    differences = []

    # Compare documents
    for doc_id, uat_doc in uat_docs.items():
        prod_doc = prod_docs.get(doc_id)
        if prod_doc:
            # Compare each field
            for field, uat_value in uat_doc.items():
                prod_value = prod_doc.get(field, None)
                if uat_value != prod_value:
                    differences.append({
                        "doc_id": doc_id,
                        "field": field,
                        "uat_value": uat_value,
                        "prod_value": prod_value
                    })
        else:
            differences.append({
                "doc_id": doc_id,
                "message": "Document missing in PROD"
            })

    # Save differences to a JSON file
    with open("field_differences.json", "w") as diff_file:
        json.dump(differences, diff_file, indent=4)

    print("Comparison complete. Differences saved to 'field_differences.json'.")
else:
    print("Failed to fetch data from one or both environments.")
