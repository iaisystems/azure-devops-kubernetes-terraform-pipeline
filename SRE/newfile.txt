from elasticsearch import Elasticsearch
import config
import requests

# --- Step 1: Get Access Token ---
token_url = f"https://{config.token_base_url}/oauth2/token"
token_response = requests.post(token_url, data={
    "grant_type": "client_credentials",
    "client_id": config.client_id,
    "client_secret": config.client_secret
})

access_token = token_response.json().get("access_token")

# --- Step 2: Connect to Elasticsearch ---
es = Elasticsearch(
    "https://your-elasticsearch-url.com",  # Replace with your actual ELK endpoint
    headers={"Authorization": f"Bearer {access_token}"}
)

# --- Step 3: Search and Fetch Only Required Fields ---
query = {
    "_source": ["a", "b", "c", "d"],  # This ensures only the needed fields are returned
    "query": {
        "match_all": {}
    }
}

response = es.search(index="bmp_issues", body=query, size=100)

# --- Step 4: Print Each Document's Selected Fields ---
print("Results:")
for hit in response["hits"]["hits"]:
    doc = hit["_source"]
    print(f"a: {doc.get('a')}, b: {doc.get('b')}, c: {doc.get('c')}, d: {doc.get('d')}")
