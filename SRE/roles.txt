class MapAitCount(luigi.Task):
    timestamp = luigi.IntParameter()

    def output(self):
        return luigi.LocalTarget(f'corpora/roles-{self.timestamp}/roles_count.json')

    def requires(self):
        return {
            'roles': MapRolesAitTask(timestamp=self.timestamp),
            'counts': VMCountByAit(timestamp=self.timestamp)
        }
