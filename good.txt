check_dup_model = api.model('DuplicateTextModel', {
    'index': fields.String(required=True, description='Elasticsearch index name'),
    'issueFilterText': fields.String(required=True, description='Text used to check for duplicate'),
    'issueCategory': fields.String(required=True, description='Issue category'),
    'issueDescription': fields.String(required=True, description='Issue description'),
    'issueSampleLog': fields.String(required=True, description='Log content')
})


from datetime import datetime
from elasticsearch import Elasticsearch, helpers

es = Elasticsearch("http://localhost:9200")

def is_duplicate_or_insert(issue_filter_text: str, index_name: str, full_doc: dict) -> dict:
    try:
        if not es.indices.exists(index=index_name):
            return {"status": "error", "message": "Index does not exist"}

        # Scan for exact match
        scan_results = helpers.scan(
            client=es,
            index=index_name,
            query={
                "query": {
                    "term": {
                        "issueFilterText.keyword": issue_filter_text
                    }
                }
            },
            size=1
        )

        for doc in scan_results:
            return {
                "status": "duplicate found",
                "id": doc['_id'],
                "matched_text": doc['_source'].get('issueFilterText')
            }

        # No duplicate — insert full document
        full_doc['issueFilterText'] = issue_filter_text
        full_doc['created_at'] = datetime.utcnow().isoformat()

        result = es.index(index=index_name, document=full_doc)
        return {
            "status": "inserted",
            "id": result['_id'],
            "inserted_text": issue_filter_text
        }

    except Exception as e:
        return {"status": "error", "message": str(e)}


@ns.route('/bmp-issue/check-duplicate-api')
class CheckDuplicateAndInsert(Resource):
    @ns.expect(check_dup_model)
    def post(self):
        payload = request.json
        index = payload.get('index')
        issue_filter_text = payload.get('issueFilterText')

        # Prepare document with all 4 fields
        doc = {
            "issueCategory": payload.get('issueCategory'),
            "issueDescription": payload.get('issueDescription'),
            "issueSampleLog": payload.get('issueSampleLog')
        }

        result = is_duplicate_or_insert(issue_filter_text, index, doc)

        if result['status'] == "error":
            return result, 500 if "Error" in result['message'] else 404
        else:
            return result, 200
