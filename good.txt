import json

# Load JSON data from two Elasticsearch responses
def load_json(filename):
    with open(filename, "r", encoding="utf-8") as f:
        return json.load(f)

# Replace these with your actual file names
file1 = "response_prod.json"  
file2 = "response_uat.json"   

# Load JSON files
data1 = load_json(file1)
data2 = load_json(file2)

# Extract the records from 'hits' -> 'hits'
data1_records = data1.get("hits", {}).get("hits", [])
data2_records = data2.get("hits", {}).get("hits", [])

# Function to compare records
def compare_records(record1, record2):
    record1 = record1.get("_source", {}) if isinstance(record1, dict) else {}
    record2 = record2.get("_source", {}) if isinstance(record2, dict) else {}

    return (
        record1.get("lastModifiedDate") == record2.get("lastModifiedDate") and
        record1.get("roleId") == record2.get("roleId")
    )

# Create two separate sections
matches_from_prod = {}
matches_from_uat = {}

# Compare and group results
for rec1 in data1_records:
    rec1_source = rec1.get("_source", {})
    matches_from_prod[str(rec1_source)] = [
        rec2["_source"] for rec2 in data2_records if compare_records(rec1, rec2)
    ]

for rec2 in data2_records:
    rec2_source = rec2.get("_source", {})
    matches_from_uat[str(rec2_source)] = [
        rec1["_source"] for rec1 in data1_records if compare_records(rec1, rec2)
    ]

# Print output in two separate sections
print("\n==============================")
print("🔹 Matches for Each Record from response_prod.json")
print("==============================\n")
print(json.dumps(matches_from_prod, indent=4))

print("\n==============================")
print("🔹 Matches for Each Record from response_uat.json")
print("==============================\n")
print(json.dumps(matches_from_uat, indent=4))
