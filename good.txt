from flask import Flask, request
from flask_restx import Api, Resource, fields
from elasticsearch import Elasticsearch

app = Flask(__name__)
api = Api(app, version='1.0', title='ELK Index Reader API',
          description='API to read Elasticsearch index data')

ns = api.namespace('elk', description='Operations on Elasticsearch')

elk_query_model = api.model('ELKQuery', {
    'index': fields.String(required=True, description='Index name'),
    'query': fields.Raw(required=False, description='Optional ES query (match_all by default)')
})

# Initialize Elasticsearch client
es = Elasticsearch("http://localhost:9200")  # Change to your ELK host and port

@ns.route('/search')
class ELKSearch(Resource):
    @ns.expect(elk_query_model)
    def post(self):
        """Search documents in a specific index"""
        payload = api.payload
        index = payload.get('index')
        query = payload.get('query', {"match_all": {}})

        try:
            if not es.indices.exists(index=index):
                return {"error": f"Index '{index}' not found"}, 404

            result = es.search(index=index, query=query)
            return {"hits": result['hits']['hits']}, 200
        except Exception as e:
            return {"error": str(e)}, 500

if __name__ == '__main__':
    app.run(debug=True)
