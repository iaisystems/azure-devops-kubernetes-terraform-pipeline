import json

# Load JSON data from files
with open("response_prod.json", "r", encoding="utf-8") as f1, open("response_uat.json", "r", encoding="utf-8") as f2:
    data1 = json.load(f1)
    data2 = json.load(f2)

# Extract actual records from the Elasticsearch response
data1_records = data1.get("hits", {}).get("hits", [])
data2_records = data2.get("hits", {}).get("hits", [])

# Function to compare records based on lastModifiedDate and roleId
def compare_records(record1, record2):
    # Extract the _source field if present
    record1 = record1.get("_source", {}) if isinstance(record1, dict) else {}
    record2 = record2.get("_source", {}) if isinstance(record2, dict) else {}

    return (
        record1.get("lastModifiedDate") == record2.get("lastModifiedDate") and
        record1.get("roleId") == record2.get("roleId")
    )

# Iterate through records and compare
matches = []
for record1 in data1_records:
    for record2 in data2_records:
        if compare_records(record1, record2):
            matches.append(record1["_source"])  # Store the matched records

# Print matching records
if matches:
    print("\n✅ Matching Records Found:\n")
    for match in matches:
        print(json.dumps(match, indent=4))
else:
    print("\n❌ No matching records found!")
